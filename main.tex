\section{Least Squares}
Consider generic setup
$$x_{t+1} = f^*(x_t) + \eta_t$$
where $f^*\in \F$, with $\F=\{f:\reals^d\to \reals^d\}$ a class of $d$-variate functions on $\reals^d$, and $\eta_t$ is a zero-mean noise vector. Suppose $n$ is time horizon and $\eta_1,\ldots,\eta_n$ are independent subGaussian. Let
$$\algo = \argmin{f\in\F} \frac{1}{n}\sum_{t=1}^n \norm{x_{t+1}-f(x_t)}^2 := \emploss(f)$$ 
be the constrained least squares solution. Optimality of $\algo$ implies
$\emploss(\algo)\leq \emploss(f^*)$, which implies
$$\sum_{t=1}^n \norm{\algo(x_t)-f^*(x_t)}^2 \leq 2\sum_{t=1}^n \inner{\eta_t, \algo(x_t)-f^*(x_t)}$$
which we write as
\begin{align}
	\sum_{t=1}^n \norm{\algo(x_t)-f^*(x_t)}^2 &\leq 4\sum_{t=1}^n \inner{\eta_t, \algo(x_t)-f^*(x_t)} - \sum_{t=1}^n \norm{\algo(x_t)-f^*(x_t)}^2 \\
	&\leq \sup_{f\in\F } \sum_{t=1}^n 4\inner{\eta_t, f(x_t)-f^*(x_t)} - \norm{f(x_t)-f^*(x_t)}^2
\end{align}
which yields sequential offset Rademacher complexity for vector-valued functions.

Also note that for any $f$, conditionally on $x_t$,
\begin{align}
	\En_{\eta_t}\norm{f(x_t)-x_{t+1}}^2 - \En_{\eta_t}\norm{f^*(x_t)-x_{t+1}}^2 = \norm{f(x_t)-f^*(x_t)}^2
\end{align}
since the cross-term vanishes. Define
$$\condemploss(f) = \frac{1}{n}\sum_{t=1}^n \En_{\eta_t}\norm{f(x_t)-x_{t+1}}^2$$
which is a random quantity. We have
$$\condemploss(f) - \condemploss(f^*) = \frac{1}{n}\sum_{t=1}^n \norm{f(x_t)-f^*(x_t)}^2$$

\section{Parametric case}

Consider $$\F = \{x\mapsto \sigma(A x): A\in \Theta\}, ~~\Theta\subseteq \reals^{d\times d}$$
and $\sigma$ is monotonic $\reals\to\reals$ applied coordinate-wise. Then
\begin{align}
	\norm{f(x_t)-f^*(x_t)}^2 = \norm{\sigma(Ax_t)-\sigma(A^* x_t)}^2 = \sum_{i=1}^d (\sigma(\inner{a_i,x_t})-\sigma(\inner{a_i^*,x_t}))^2
\end{align}
Observe that
$$\sigma(\inner{a_i,x_t})-\sigma(\inner{a_i^*,x_t}) = \inner{a_i-a_i^*,x_t} \sigma'(\xi)$$
for $\xi\in[\inner{a_i,x_t}, \inner{a_i^*,x_t}]$
and since $\sigma$ is nondecreasing, the product 
$$(\sigma(\inner{a_i,x_t})-\sigma(\inner{a_i^*,x_t}) )\cdot \inner{a_i-a_i^*,x_t} \geq 0.$$
This means that
\begin{align}
	(\sigma(\inner{a_i,x_t})-\sigma(\inner{a_i^*,x_t}))^2 &= (\sigma(\inner{a_i,x_t})-\sigma(\inner{a_i^*,x_t})) \cdot \inner{a_i-a_i^*,x_t}) \cdot \sigma'(\xi)\\
	&\leq (\sigma(\inner{a_i,x_t})-\sigma(\inner{a_i^*,x_t})) \cdot \inner{a_i-a_i^*,x_t}) \cdot L
\end{align}
where $|\sigma'|_\infty\leq L$. We conclude that
\begin{align}
	\condemploss(A)-\condemploss(A^*) &= \sum_{i=1}^d (\sigma(\inner{a_i,x_t})-\sigma(\inner{a_i^*,x_t}))^2\\
	&\leq L \sum_{i=1}^d (\sigma(\inner{a_i,x_t})-\sigma(\inner{a_i^*,x_t})) \cdot \inner{a_i-a_i^*,x_t}\\
	&= \inner{g(A,A^*), A-A^*}
\end{align}
with $g(A,A^*)$ is a matrix whose $i$th row is $[g(A,A^*)]_i = (\sigma(\inner{a_i,x_t})-\sigma(\inner{a_i^*,x_t})) x_t$. 

Furthermore, since
\begin{align}
	\nabla  (\sigma(\inner{a_i,x_t})-\sigma(\inner{a_i^*,x_t}))^2 
	&=(\sigma(\inner{a_i,x_t})-\sigma(\inner{a_i^*,x_t})) \cdot \sigma'(\inner{a_i,x_t}) \cdot x_t 
\end{align}
we can write
\begin{align}
	\condemploss(A)-\condemploss(A^*) 
	&\leq L \sum_{i=1}^d (\sigma(\inner{a_i,x_t})-\sigma(\inner{a_i^*,x_t})) \cdot \inner{a_i-a_i^*,x_t}\\
	&\leq (L/\ell) \sum_{i=1}^d (\sigma(\inner{a_i,x_t})-\sigma(\inner{a_i^*,x_t})) \cdot \inner{a_i-a_i^*,x_t} \cdot \sigma'(\inner{a_i,x_t})\\
	&= (L/\ell) \inner{\nabla \condemploss(A), A-A^*}
\end{align}
and $\ell \leq |\sigma'|_\infty$.

\section{$\condemploss$ vs $\emploss$}
\begin{align}
	\En\sup_{f\in\F}\left[\condemploss-\emploss\right] &= \En\sup_{f\in\F}\left[\frac{1}{n}\sum_{t=1}^n \En_{\eta_t}\norm{x_{t+1}-f(x_t)}^2 - \frac{1}{n}\sum_{t=1}^n \norm{x_{t+1}-f(x_t)}^2 \right]\\
	&=\En\sup_{f\in\F}\left[\frac{1}{n}\sum_{t=1}^n 2\inner{\En_{\eta_t}[x_{t+1}']-x_{t+1},f(x_t)}\right] + \En\left[\frac{1}{n}\sum_{t=1}^n \En_{\eta_t}\norm{x_{t+1}'}^2-\norm{x_{t+1}}^2 \right]
\end{align}

\section{Ideas}
\begin{itemize}
\item Prove restricted eigenvalue for the empirical design matrix to get $s\log(d)/n$-type rates for the sparse setting.
\item Prove covering number bounds for generic dynamical systems when $p\geq{}2$ using the skeleton aggregation technique adapted to this sequential setting (we have to figure out how to avoid sample splitting).
\item Get efficient algorithms for the case where $\sigma$ is Lipschitz but unknown via isotron-style approach.
\begin{itemize}
\item Another note: Can we get the parametric rate for estimating $\Theta$ in this unknown $\sigma$ setting? Eg, using semiparametric approaches? One guess: If it's the same $\sigma$ across all coordinates, then the orthogonal/double machine learning-type approach may actually suffice. 
\item Basic idea: If we were in the iid setting we could sample split and learn $\sigma$ using isotron/PAV, then use semiparametric least squares to learn $\Theta$. How does this argument work in the dependent setting?
\item Can having large dimension actually make things easier here, since it means we get more "measurements" per time step (assuming $\sigma$ is fixed across all coordinates).
\end{itemize}
\item Handle the case where $\sigma(x)=\mathrm{sgn}(x)$. To show that this works for least squares using VC-type arguments, we basically need to show that our tree is not worst-case.
\item Minimax lower bounds.
\end{itemize}
