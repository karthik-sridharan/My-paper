\section{A Result}
\begin{theorem}[Jensen's Inequality]
Let \( F: \mathcal{X} \to \mathbb{R} \) be a convex function on a set \( \mathcal{X} \subseteq \mathbb{R} \), and let \( X \) be a random variable taking values in \( \mathcal{X} \) with finite expectation. Then
\[
F\big(\mathbb{E}[X]\big) \leq \mathbb{E}[F(X)].
\]
\end{theorem}

\begin{proof}
By the definition of convexity for vector-valued arguments, for any \( \mathbf{x}_0 \in I \subseteq \mathbb{R}^n \), there exists a subgradient (gradient) \( \nabla F(\mathbf{x}_0) \in \mathbb{R}^n \) such that
\[
F(\mathbf{x}) \geq F(\mathbf{x}_0) + \langle \nabla F(\mathbf{x}_0), \mathbf{x} - \mathbf{x}_0 \rangle \quad \text{for all } \mathbf{x} \in I.
\]
Set \( \mathbf{x}_0 = \mathbb{E}[\mathbf{X}] \). Taking expectation on both sides,
\[
\mathbb{E}[F(\mathbf{X})] \geq F(\mathbb{E}[\mathbf{X}]) + \mathbb{E}\big[\langle \nabla F(\mathbb{E}[\mathbf{X}]), \mathbf{X} - \mathbb{E}[\mathbf{X}] \rangle\big] = F(\mathbb{E}[\mathbf{X}]) + \langle \nabla F(\mathbb{E}[\mathbf{X}]), \mathbb{E}[\mathbf{X}] - \mathbb{E}[\mathbf{X}] \rangle = F(\mathbb{E}[\mathbf{X}]).
\]
This completes the proof.
\end{proof}
