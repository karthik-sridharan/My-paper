\begin{frame}{Jensen's Inequality}
Statement of Jensen's Inequality for convex functions and random vectors

\begin{block}{Theorem (Jensen's Inequality)}
Let \( F: \mathcal{X} \to \mathbb{R} \) be convex on convex set \( \mathcal{X} \subseteq \mathbb{R}^n \), and \( \mathbf{X} \) a random vector with \( \mathbb{E}[\mathbf{X}] \in \mathcal{X} \). Then:
\[ F\big(\mathbb{E}[\mathbf{X}]\big) \leq \mathbb{E}[F(\mathbf{X})]. \]
\end{block}
\end{frame}

\begin{frame}{Proof Sketch}
Using subgradients and linearity of expectation

\begin{block}{Proof Outline}
By convexity, for any \( \mathbf{x}_0 \in \mathcal{X} \), there exists \( \mathbf{g} \in \partial F(\mathbf{x}_0) \) with:
\[ F(\mathbf{x}) \geq F(\mathbf{x}_0) + \langle \mathbf{g}, \mathbf{x} - \mathbf{x}_0 \rangle, \quad \forall \mathbf{x} \in \mathcal{X}. \]
Set \( \mathbf{x}_0 = \mathbb{E}[\mathbf{X}] \) and take expectation:
\[ \mathbb{E}[F(\mathbf{X})] \geq F(\mathbb{E}[\mathbf{X}]) + \langle \mathbf{g}, \mathbb{E}[\mathbf{X}] - \mathbb{E}[\mathbf{X}] \rangle = F(\mathbb{E}[\mathbf{X}]). \]
\end{block}
\end{frame}

\begin{frame}{Intuition Behind Jensen's Inequality}
Convex functions lie below their tangent lines

\begin{block}{Key Idea}
\begin{itemize}
\item Convexity means function lies below any supporting hyperplane
\item Expectation is a weighted average of random variable values
\item Inequality shows function of average \( \leq \) average of function values
\item Visualized as chord lying above curve for convex functions
\end{itemize}
\end{block}
\end{frame}
